function _qlInvokeBridge(payload) {
  const apiKey = (PropertiesService.getScriptProperties().getProperty('RAPIDAPI_KEY') || '').trim();
  const rawJson = JSON.stringify(payload);
  const digest = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, (apiKey ? 'PRO:' : 'DEMO:') + rawJson);
  const cacheKey = 'ql_' + digest.slice(0, 16).map(function (b) { return ('0' + (b & 0xff).toString(16)).slice(-2); }).join('');
  const cache = CacheService.getScriptCache();
  const hit = cache.get(cacheKey);
  if (hit) {
    return JSON.parse(hit);
  }
  const url = apiKey ? QL_RAPIDAPI_URL : QL_FREE_DEMO_URL;
  const headers = { 'Content-Type': 'application/json' };
  if (apiKey) {
    headers['X-RapidAPI-Key'] = apiKey;
    headers['X-RapidAPI-Host'] = QL_RAPIDAPI_HOST;
  }
  const resp = UrlFetchApp.fetch(url, {
    method: 'post',
    headers: headers,
    payload: rawJson,
    muteHttpExceptions: true,
  });
  const code = resp.getResponseCode();
  const data = JSON.parse(resp.getContentText() || '{}');
  if (code >= 400) {
    throw new Error(data.detail || data.message || ('QuantLedger HTTP ' + code));
  }
  const out = (String(payload.output_mode || '').toLowerCase().indexOf('scalar') === 0)
    ? data.scalar_value
    : (data.rows || [['No rows returned']]);
  const serialized = JSON.stringify(out);
  if (serialized.length < 95000) {
    cache.put(cacheKey, serialized, 600);
  }
  return out;
}

/**
 * Executes any of QuantLedger's 73 deterministic GAAP, IRC Tax, 409A, QoE, DCAA, CRE or Credit engines (BI-01..BI-73).
 * @param {"BI-17"} engineCode Engine code from BI-01 to BI-73 (e.g. "BI-17" ASC 842 Lease, "BI-07" DCAA Wrap, "BI-48" 409A).
 * @param {"NY"} jurisdiction 2-letter US State, DC, PR, GU, VI, AS, MP, or US.
 * @param {A2:B8} inputRange Optional 2-column [parameter_name, value] cell range or JSON string.
 * @param {"kpis"} outputMode "kpis" (2-col summary), "schedule" (multi-period tie-out table), or "scalar:<metric_key>".
 * @return 2D spill array or scalar value.
 * @customfunction
 */
function QUANTLEDGER(engineCode, jurisdiction, inputRange, outputMode) {
  let parsedInputs = inputRange || null;
  if (typeof inputRange === 'string' && inputRange.trim().charAt(0) === '{') {
    parsedInputs = JSON.parse(inputRange);
  }
  return _qlInvokeBridge({
    formula_type: 'ENGINE',
    engine_code: engineCode || 'BI-17',
    jurisdiction: jurisdiction || 'DC',
    output_mode: outputMode || 'kpis',
    range_or_inputs: parsedInputs,
  });
}

/**
 * QuantLedger Microsoft Excel Automate (Office Script) — 2D Statutory & Compliance Spill Writer
 * Version: 2.8.2 | Docs: https://quantledgerapi.github.io/google-sheets-excel-formulas-and-n8n/
 *
 * HOW TO USE IN EXCEL (WEB OR DESKTOP):
 *   1. Open Excel -> Automate tab -> New Script -> paste this TypeScript file and click Save.
 *   2. Click Run (works immediately in free 12-row demo mode, or paste your RapidAPI key below for 100-row Pro mode).
 */

async function main(
  workbook: ExcelScript.Workbook,
  formulaType: string = "ENGINE",
  engineCode: string = "BI-17",
  jurisdiction: string = "NY",
  outputMode: string = "schedule",
  rapidApiKey: string = ""
): Promise<void> {
  const sheet = workbook.getActiveWorksheet();
  const usePro = rapidApiKey.trim().length > 0;
  const endpoint = usePro
    ? "https://quantledger-73-engine-financial-and-statutory-api.p.rapidapi.com/v1/spreadsheet/formula-eval"
    : "https://quantledger-ghost-kitchen-3yplwqtnha-uc.a.run.app/v1/public/spreadsheet-demo";

  const headers: Record<string, string> = { "Content-Type": "application/json" };
  if (usePro) {
    headers["X-RapidAPI-Key"] = rapidApiKey.trim();
    headers["X-RapidAPI-Host"] = "quantledger-73-engine-financial-and-statutory-api.p.rapidapi.com";
  }

  const response = await fetch(endpoint, {
    method: "POST",
    headers,
    body: JSON.stringify({
      formula_type: formulaType,
      engine_code: engineCode,
      jurisdiction,
      output_mode: outputMode,
      max_rows: usePro ? 50 : 12,
    }),
  });

  const payload = await response.json();
  if (!response.ok) {
    throw new Error(payload?.detail || `QuantLedger HTTP ${response.status}`);
  }

  const rows: (string | number | boolean)[][] = payload.rows || [["No data"]];
  const rowCount = rows.length;
  const colCount = rows[0].length;

  const targetRange = sheet.getRangeByIndexes(0, 0, rowCount, colCount);
  targetRange.setValues(rows);
  const headerRange = sheet.getRangeByIndexes(0, 0, 1, colCount);
  headerRange.getFormat().getFill().setColor("#131B33");
  headerRange.getFormat().getFont().setColor("#FFFFFF");
  headerRange.getFormat().getFont().setBold(true);
  targetRange.getFormat().autofitColumns();
}

// QuantLedger Microsoft Excel Desktop Power Query (M) Function — v2.8.2
// 1. In Excel Desktop: Data -> Get Data -> From Other Sources -> Blank Query -> Advanced Editor
// 2. Paste this query, click Done, and invoke fnQuantLedger("ENGINE", "BI-17", "NY", "schedule", "")
let
    fnQuantLedger = (
        formulaType as text,
        engineCode as text,
        jurisdiction as text,
        outputMode as text,
        optional rapidApiKey as nullable text
    ) as table =>
    let
        cleanKey = if rapidApiKey = null then "" else Text.Trim(rapidApiKey),
        usePro = Text.Length(cleanKey) > 0,
        targetUrl = if usePro
            then "https://quantledger-73-engine-financial-and-statutory-api.p.rapidapi.com/v1/spreadsheet/formula-eval"
            else "https://quantledger-ghost-kitchen-3yplwqtnha-uc.a.run.app/v1/public/spreadsheet-demo",
        reqHeaders = if usePro
            then [#"Content-Type"="application/json", #"X-RapidAPI-Key"=cleanKey, #"X-RapidAPI-Host"="quantledger-73-engine-financial-and-statutory-api.p.rapidapi.com"]
            else [#"Content-Type"="application/json"],
        bodyBytes = Json.FromValue([
            formula_type = formulaType,
            engine_code = engineCode,
            jurisdiction = jurisdiction,
            output_mode = outputMode,
            max_rows = if usePro then 100 else 12
        ]),
        rawResp = Web.Contents(targetUrl, [Headers = reqHeaders, Content = bodyBytes]),
        parsed = Json.Document(rawResp),
        rowsList = parsed[rows],
        colNames = List.Transform(rowsList{0}, each Text.From(_)),
        dataRows = List.Skip(rowsList, 1),
        outTable = Table.FromRows(dataRows, colNames)
    in
        outTable
in
    fnQuantLedger
